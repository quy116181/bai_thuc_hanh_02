//Bài 1
#include <stdio.h>
#include <stdlib.h>

int main() {
    int n;
    printf("Nhap so phan tu n: ");
    scanf("%d", &n);

  
    float *a = (float *)malloc(n * sizeof(float));
    if (a == NULL) {
        printf("Khong du bo nho!\n");
        return 1;
    }

  
    printf("Nhap cac phan tu cua mang:\n");
    for (int i = 0; i < n; i++) {
        printf("a[%d] = ", i);
        scanf("%f", (a + i)); 
    }

    
    printf("\nMang vua nhap la: ");
    for (int i = 0; i < n; i++) {
        printf("%.2f ", *(a + i));
    }
    printf("\n");

    
    free(a);
    return 0;
}
